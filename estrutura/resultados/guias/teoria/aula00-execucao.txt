tokenizar <- function(texto) {     # function(argumentos) { corpo }
  texto <- tolower(texto)          # minúsculas: "Modelo" e "modelo" viram o mesmo
  unlist(strsplit(texto, " "))     # a ÚLTIMA expressão é o valor devolvido
}                                  # fim da função
